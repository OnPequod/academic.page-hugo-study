{{- /* Downloads the site's publications as one BibTeX file and publishes it at /publications/publications.bib. Returns the published resource, or nil. */ -}} {{- $cfg := partialCached "api/config.html" "config" -}} {{- $url := printf "%s/publications.bib" $cfg.root -}} {{- $headers := merge $cfg.headers (dict "Accept" "application/x-bibtex, application/json") -}} {{- $out := false -}} {{- with try (resources.GetRemote $url (dict "headers" $headers)) -}} {{- with .Err -}} {{- errorf "Academic.page API request to %s failed: %s" $url . -}} {{- else with .Value -}} {{- if .Content -}} {{- $out = resources.Copy "publications/publications.bib" . -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
