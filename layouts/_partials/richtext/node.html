{{- /* Renders one Tiptap node and its children to HTML. Supported nodes: doc, paragraph, heading (1-3), text, hardBreak, bulletList, orderedList, listItem, blockquote, codeBlock, horizontalRule, and image (resolved through the media library when it has a data-media-id). Supported marks: bold, italic, strike, code, and link. Other nodes render their children only, so their text survives and their markup does not. Text is always escaped, and links keep only http, https, mailto, site-relative, and fragment URLs. */ -}} {{- $n := . -}} {{- $inner := "" -}} {{- range $n.content -}}{{- $inner = printf "%s%s" $inner (partial "richtext/node.html" .) -}}{{- end -}} {{- $attrs := $n.attrs | default dict -}} {{- $t := $n.type -}} {{- $h := "" -}} {{- if eq $t "text" -}} {{- $h = htmlEscape ($n.text | default "") -}} {{- $open := "" -}} {{- $close := "" -}} {{- range $n.marks -}} {{- $tag := "" -}} {{- $attr := "" -}} {{- if eq .type "bold" -}}{{- $tag = "strong" -}} {{- else if eq .type "italic" -}}{{- $tag = "em" -}} {{- else if eq .type "strike" -}}{{- $tag = "s" -}} {{- else if eq .type "code" -}}{{- $tag = "code" -}} {{- else if eq .type "link" -}} {{- $href := (.attrs | default dict).href | default "" -}} {{- if findRE `^(?i)(https?:|mailto:|/|#)` $href -}} {{- $tag = "a" -}} {{- $attr = printf ` href="%s"` (htmlEscape $href) -}} {{- end -}} {{- end -}} {{- with $tag -}} {{- $open = printf "%s<%s%s>" $open . $attr -}} {{- $close = printf "%s" . $close -}} {{- end -}} {{- end -}} {{- $h = printf "%s%s%s" $open $h $close -}} {{- else if eq $t "doc" -}} {{- $h = $inner -}} {{- else if eq $t "paragraph" -}} {{- $h = printf "

%s

" $inner -}} {{- else if eq $t "heading" -}} {{- $level := int ($attrs.level | default 2) -}} {{- if or (lt $level 1) (gt $level 3) -}}{{- $level = 3 -}}{{- end -}} {{- $h = printf "%s" $level $inner $level -}} {{- else if eq $t "hardBreak" -}} {{- $h = "
" -}} {{- else if eq $t "bulletList" -}} {{- $h = printf "
    %s
" $inner -}} {{- else if eq $t "orderedList" -}} {{- $start := int ($attrs.start | default 1) -}} {{- $h = cond (eq $start 1) (printf "
    %s
" $inner) (printf `
    %s
` $start $inner) -}} {{- else if eq $t "listItem" -}} {{- $h = printf "
  • %s
    • " $inner -}} {{- else if eq $t "blockquote" -}} {{- $h = printf "
    %s
    " $inner -}} {{- else if eq $t "codeBlock" -}} {{- $lang := "" -}} {{- with findRE `^[A-Za-z0-9_+#-]+$` ($attrs.language | default "") -}}{{- $lang = index . 0 -}}{{- end -}} {{- $class := cond (ne $lang "") (printf ` class="language-%s"` $lang) "" -}} {{- $h = printf "
    %s
    " $class $inner -}} {{- else if eq $t "horizontalRule" -}} {{- $h = "
    " -}} {{- else if eq $t "image" -}} {{- $h = partial "richtext/image.html" $attrs -}} {{- else -}} {{- $h = $inner -}} {{- end -}} {{- return $h -}}
